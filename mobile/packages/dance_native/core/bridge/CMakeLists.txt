cmake_minimum_required(VERSION 3.18)
project(woah_privacy_core_bridge LANGUAGES C)
set(CMAKE_C_STANDARD 11)
if(NOT DEFINED WOAH_RUST_LIBRARY)
  message(FATAL_ERROR "WOAH_RUST_LIBRARY must identify the compiled Rust static library")
endif()
add_library(woah_privacy_core_jni SHARED jni.c)
target_include_directories(woah_privacy_core_jni PRIVATE ../include)
target_link_libraries(woah_privacy_core_jni PRIVATE "${WOAH_RUST_LIBRARY}")
if(ANDROID)
  target_link_libraries(woah_privacy_core_jni PRIVATE dl m)
  target_link_options(woah_privacy_core_jni PRIVATE "-Wl,-z,max-page-size=16384")
else()
  find_package(JNI REQUIRED)
  target_include_directories(woah_privacy_core_jni PRIVATE ${JNI_INCLUDE_DIRS})
  if(WIN32)
    target_link_libraries(woah_privacy_core_jni PRIVATE ws2_32 userenv bcrypt ntdll advapi32)
  elseif(APPLE)
    target_link_libraries(woah_privacy_core_jni PRIVATE iconv)
  else()
    target_link_libraries(woah_privacy_core_jni PRIVATE dl pthread m rt util)
  endif()
endif()
