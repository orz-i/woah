cmake_minimum_required(VERSION 3.18)
project(frame_kernel_bridge LANGUAGES C)
set(CMAKE_C_STANDARD 11)
if(NOT DEFINED FRAME_KERNEL_LIBRARY)
  message(FATAL_ERROR "FRAME_KERNEL_LIBRARY must identify the compiled Rust static library")
endif()
add_library(frame_kernel_jni SHARED jni.c)
target_include_directories(frame_kernel_jni PRIVATE "${FRAME_KERNEL_INCLUDE}")
target_link_libraries(frame_kernel_jni PRIVATE "${FRAME_KERNEL_LIBRARY}")
if(ANDROID)
  target_link_libraries(frame_kernel_jni PRIVATE dl m)
  target_link_options(frame_kernel_jni PRIVATE "-Wl,-z,max-page-size=16384")
else()
  find_package(JNI REQUIRED)
  target_include_directories(frame_kernel_jni PRIVATE ${JNI_INCLUDE_DIRS})
  if(WIN32)
    target_link_libraries(frame_kernel_jni PRIVATE ws2_32 userenv bcrypt ntdll advapi32)
  elseif(APPLE)
    target_link_libraries(frame_kernel_jni PRIVATE iconv)
  else()
    target_link_libraries(frame_kernel_jni PRIVATE dl pthread m rt util)
  endif()
endif()
